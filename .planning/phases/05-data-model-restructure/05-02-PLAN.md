```typescript
export function resolveItem(id: string): ResolvedItem | null {
  // 先查 supplement
  const supplement = SUPPLEMENT_MAP.get(id);
  if (supplement) {
    return {
      id: supplement.id,
      type: "supplement",
      name: supplement.name,
      dose: supplement.dosagePerUnit,
      cal: 0,
      tags: supplement.tags,
      description: supplement.mechanism ?? "",
      tcm: supplement.tcm,
      caution: supplement.caution,
      isCore: supplement.isActive,
      raw: supplement,
    };
  }

  // 再查 food
  const food = FOOD_MAP.get(id);
  if (food) {
    return {
      id: food.id,
      type: "food",
      name: food.name,
      dose: food.serving,
      cal: food.cal,
      tags: food.tags ?? [],
      description: `P${food.protein}g / F${food.fat}g / C${food.carbs}g / Na${food.sodium}mg`,
      isCore: false,
      raw: food,
    };
  }

  console.warn(`[resolveItem] Unknown ID: ${id}`);
  return null;
}
```

Key mapping changes from old RemedyItem to SupplementItem:
- `r.dose` -> `supplement.dosagePerUnit` (dose field no longer exists)
- `r.cal ?? 0` -> `0` (SupplementItem has no cal field)
- `r.mechanism` -> `supplement.mechanism ?? ""` (now optional)
- `r.isCore ?? false` -> `supplement.isActive` (isCore replaced by isActive)
