   ```typescript
   function handleSubmit() {
     // Validate
     if (!name.trim()) { setNameError("請輸入食材名稱"); return; }
     if (!serving.trim()) { setServingError("請輸入份量"); return; }

     const foodItem: FoodItem = {
       id: food?.id ?? `food_${Date.now()}`,  // Edit reuses ID, add generates new
       type: "food",
       name: name.trim(),
       serving: serving.trim(),
       cal: parseFloat(cal) || 0,
       protein: parseFloat(protein) || 0,
       fat: parseFloat(fat) || 0,
       carbs: parseFloat(carbs) || 0,
       sugar: sugar.trim() ? parseFloat(sugar) : undefined,
       sodium: parseFloat(sodium) || 0,
       source: source.trim(),
       tags: tags.length > 0 ? tags : undefined,
       ingredients: food?.ingredients,  // Preserve existing ingredients if editing a composed food
     };
     onSave(foodItem);
   }
   ```
