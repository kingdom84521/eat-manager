4. Add four new methods to the ItemService singleton object:

   **logConsumption** (per D-12, D-19 — append-only event sourcing, same pattern as `upsertInventory`):
   ```typescript
   async logConsumption(event: ConsumptionEvent): Promise<void> {
     const existing = cacheGet<ConsumptionEvent[]>(CACHE_KEYS.CONSUMPTION) ?? [];
     existing.push(event);
     cacheSet(CACHE_KEYS.CONSUMPTION, existing);
     SheetsAPI.append(SHEETS.CONSUMPTION, event as unknown as SheetRow).catch(() => {});
   },
   ```

   **getConsumption** (per D-20 — reads from localStorage, background syncs from Sheets):
   ```typescript
   async getConsumption(supplementId?: string): Promise<ConsumptionEvent[]> {
     const cached = cacheGet<ConsumptionEvent[]>(CACHE_KEYS.CONSUMPTION) ?? [];
     SheetsAPI.readAll(SHEETS.CONSUMPTION)
       .then((rows) => {
         if (rows.length > 0) {
           cacheSet(CACHE_KEYS.CONSUMPTION, rows.map(rowToConsumption));
         }
       })
       .catch(() => {});
     if (supplementId) return cached.filter((e) => e.supplementId === supplementId);
     return cached;
   },
   ```

   **getDailyLog** (per D-14 — localStorage only, no Sheets sync):
   ```typescript
   getDailyLog(date: string): SupplementLogEntry | null {
     return cacheGet<SupplementLogEntry>(CACHE_KEYS.SUPPLEMENT_LOG + "_" + date);
   },
   ```

   **saveDailyLog** (per D-14 — localStorage only, no Sheets sync):
   ```typescript
   saveDailyLog(entry: SupplementLogEntry): void {
     cacheSet(CACHE_KEYS.SUPPLEMENT_LOG + "_" + entry.date, entry);
   },
   ```

Note: `getDailyLog` and `saveDailyLog` are synchronous (no `async`) because they only touch localStorage — no network calls. This differs from the other methods. The research recommended localStorage-only for SupplementLogEntry (Open Question 1 answer).

   ```typescript
   function calcRemainingUnits(suppId: string, inv: InventoryEntry[], consumption: ConsumptionEvent[]): number {
     const purchased = inv.filter((e) => e.supplementId === suppId).reduce((sum, e) => sum + e.purchasedUnits, 0);
     const consumed = consumption.filter((e) => e.supplementId === suppId).reduce((sum, e) => sum + e.units, 0);

   }
   ```

3. Update `calcDaysRemaining` function to also accept and use consumption:
   ```typescript
   function calcDaysRemaining(
     suppId: string,
     unitsPerDose: number,
     dosesPerDay: number,
     inv: InventoryEntry[],
     consumption: ConsumptionEvent[]
   ): number | null {
     const entries = inv.filter((e) => e.supplementId === suppId);
     if (entries.length === 0) return null;
     const remaining = calcRemainingUnits(suppId, inv, consumption);
     const dailyUsage = unitsPerDose * dosesPerDay;
     if (dailyUsage <= 0) return null;
     return remaining / dailyUsage;
   }
   ```
