```typescript
/** 依 id 更新菜單預設的完整內容 */
update(preset: MenuPreset): void {
  const existing = this.getAll();
  const idx = existing.findIndex((p) => p.id === preset.id);
  if (idx >= 0) {
    existing[idx] = preset;
  } else {
    existing.unshift(preset);
  }
  cacheSet(MENU_KEY, existing);
},
```

This upserts by `id`: if found, replaces in-place; if not found, prepends (fallback for edge cases). Uses the same `cacheGet`/`cacheSet` helpers and `MENU_KEY` constant already in the file.
